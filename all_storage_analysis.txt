    @RequiresApi(Build.VERSION_CODES.O)
    private fun showYearMonthPicker() {

        dialog.setContentView(view)

        val npYear = view.findViewById<NumberPicker>(R.id.np_year)
        val npMonth = view.findViewById<NumberPicker>(R.id.np_month)
        val btnConfirm = view.findViewById<Button>(R.id.btn_confirm)

        val currentPosition = calendarViewPager.currentItem
        val currentTarget = getTargetMonthForPosition(currentPosition)

        npYear.minValue = 2000
        npYear.maxValue = 2099
        npYear.value = currentTarget.year
        npYear.wrapSelectorWheel = false

        npMonth.minValue = 1
        npMonth.maxValue = 12
        npMonth.value = currentTarget.monthValue
        npMonth.wrapSelectorWheel = true

        btnConfirm.setOnClickListener {
            val selectedYear = npYear.value
            val selectedMonth = npMonth.value

            val targetYearMonth = YearMonth.of(selectedYear, selectedMonth)
            val baseYearMonth = YearMonth.now()

            val monthDiff = ChronoUnit.MONTHS.between(baseYearMonth, targetYearMonth)
            val newPosition = CENTER_POSITION + monthDiff.toInt()

            calendarViewPager.setCurrentItem(newPosition, false)
            dialog.dismiss()
        }

        dialog.show()
    }

    private fun setupSwipeUpGesture() {
        val detector = GestureDetectorCompat(this,
            object : GestureDetector.SimpleOnGestureListener() {
                override fun onFling(
                    e1: MotionEvent?, e2: MotionEvent,
                    velocityX: Float, velocityY: Float
                ): Boolean {
                    val start = e1 ?: return false
                    val dy = e2.y - start.y
                    val dx = e2.x - start.x
                    if (dy < -100f && abs(dy) > abs(dx) * 1.5f && velocityY < -300f) {
                        startActivity(Intent(this@CalendarActivity, StatisticsWidgetActivity::class.java))
                        overridePendingTransition(R.anim.slide_in_up, R.anim.slide_out_up)
                        return true
                    }
                    return false
                }

                override fun onDown(e: MotionEvent) = true
            })

        rootLayout.setOnTouchListener { v, event ->
            val consumed = detector.onTouchEvent(event)
            if (!consumed) v.performClick()
            consumed
        }
    }

    private fun setupSwipeUpHint() {
        val pill = findViewById<View>(R.id.swipe_up_hint_pill)
        swipeUpAnimator = ObjectAnimator.ofFloat(pill, "translationY", 0f, -24f).apply {
            duration = 600
            repeatCount = ObjectAnimator.INFINITE
            repeatMode = ObjectAnimator.REVERSE
            interpolator = android.view.animation.AccelerateDecelerateInterpolator()
        }
        pill.postDelayed({ swipeUpAnimator.start() }, 500)
    }

            .putString(key(), rewardId)
            .apply()
    }

            .getString(key(), null)

    fun clear(context: Context, rewardId: String) {
        val prefs = context.getSharedPreferences(PREF_NAME, Context.MODE_PRIVATE)
        if (prefs.getString(key(), null) == rewardId) {

        }
    }

data class ConsumeItemsRequest(val requestId: String, val itemIds: List<Int>)
data class InventoryCount(val itemId: Int, val quantity: Int)
data class ConsumeItemsResponse(val duplicate: Boolean, val items: List<InventoryCount>)
private data class PendingAlchemy(val request: ConsumeItemsRequest, val fox: CompleteFox)

/** Persist the recipe AND its resulting fox before spending materials.
 * Recovery uses the same request/fox ID, including after process death.
 */
class AlchemyRepository(context: Context) {
    private val context = context.applicationContext
    private val prefs = this.context.getSharedPreferences("pending_alchemy", Context.MODE_PRIVATE)
    private val gson = Gson()

    fun hasPending(): Boolean = prefs.contains("user_${SessionManager.getUserId()}")

    suspend fun complete(createDraft: ((Int) -> CompleteFox)? = null): CompleteFox = operationMutex.withLock {
        val userId = SessionManager.getUserId()
        val token = SessionManager.getToken()
        check(userId >= 0 && token != null) { "로그인이 필요합니다." }
        val key = "user_$userId"
        val pending = prefs.getString(key, null)?.let { gson.fromJson(it, PendingAlchemy::class.java) }
            ?: run {
                val fox = withContext(Dispatchers.Main.immediate) {
                    checkAccount(userId)
                    // Reserve an ID across unfinished drafts too (including another test account).
                    val reservedIds = prefs.all.values.filterIsInstance<String>().mapNotNull {
                        runCatching { gson.fromJson(it, PendingAlchemy::class.java).fox.id }.getOrNull()
                    }
                    val nextId = Math.addExact((AllFoxMap.allFoxMap.keys + reservedIds).maxOrNull() ?: 0, 1)
                    checkNotNull(createDraft) { "확인할 조합이 없습니다." }.invoke(nextId)
                }
                val ids = listOfNotNull(fox.glasses, fox.hat, fox.scarf, fox.accessory).sorted()
                check(ids.isNotEmpty()) { "아이템을 선택해 주세요." }
                PendingAlchemy(ConsumeItemsRequest(UUID.randomUUID().toString(), ids), fox).also {
                    withContext(Dispatchers.IO) {
                        check(prefs.edit().putString(key, gson.toJson(it)).commit()) { "조합 요청을 저장하지 못했습니다." }
                    }
                }
            }
        checkAccount(userId)
        val response = ApiClient.api.consumeShopItems("Bearer $token", pending.request)
        checkAccount(userId)
        if (!response.isSuccessful) {
            val body = runCatching { gson.fromJson(response.errorBody()?.string(), JsonObject::class.java) }.getOrNull()
            val code = body?.get("code")?.asString
            // Only an explicit final rejection permits a new request. Timeouts/auth/5xx keep the draft.
            if (code in listOf("INSUFFICIENT_ITEMS", "INVALID_CONSUMPTION", "IDEMPOTENCY_KEY_CONFLICT")) {
                withContext(Dispatchers.IO) {
                    check(prefs.edit().remove(key).commit()) { "조합 결과를 저장하지 못했습니다." }
                }
            }
            error(body?.get("message")?.asString ?: "조합 결과를 확인하지 못했어요. 솥을 눌러 다시 확인해 주세요.")
        }
        val result = response.body()?.data ?: error("조합 결과가 없습니다. 다시 확인해 주세요.")
        withContext(Dispatchers.Main.immediate) {
            checkAccount(userId)
            ItemPrefs.replaceCounts(context, userId, result.items.associate { it.itemId to it.quantity })
            // Durable local save first; reapplying this fox after a crash replaces the same ID.
            FoxPrefs.save(context, pending.fox, durable = true)
        }
        withContext(Dispatchers.IO) {
            check(prefs.edit().remove(key).commit()) { "조합 완료 기록을 저장하지 못했습니다. 다시 확인해 주세요." }
        }
        checkAccount(userId)
        pending.fox
    }

    private fun checkAccount(userId: Int) {
        check(SessionManager.getUserId() == userId) { "로그인 계정이 변경되었습니다." }
    }

    private companion object { val operationMutex = Mutex() }
}

package com.example.sumdays.alchemy

import android.content.Context
import com.example.sumdays.auth.AccountStorage

/** Selection is UI state, not inventory. Cancel/rotation never refunds server items. */
object AlchemySelectionManager {
    private const val PREF_NAME = "alchemy_selection"
    private const val KEY_SELECTED_IDS = "selected_item_ids"

    private var owner: String? = null
    private val selected = mutableMapOf<ItemCategory, FoxShopItem>()

    private fun checkOwner() {
        val current = AccountStorage.accountKey()
        if (current != owner) {
            selected.clear()
            owner = current
        }
    }

    /** Drop the previous account's in-memory draft after a session change. */
    fun resetMemory() {
        selected.clear()
        owner = null
    }

    fun toggle(context: Context, item: FoxShopItem): List<FoxShopItem> {
        checkOwner()
        if (selected[item.itemCategory]?.id == item.id) selected.remove(item.itemCategory)
        else if (ItemPrefs.getCount(context, item.id) > 0) selected[item.itemCategory] = item
        saveSelection(context)
        return getSelectedItems()
    }

    /** Restore draft selection after recreation without reserving or decrementing inventory. */
    fun resumePendingSelection(context: Context) {
        checkOwner()
        removeLegacyDraft(context)
        val ids = AccountStorage.preferences(context, PREF_NAME)
            .getStringSet(KEY_SELECTED_IDS, emptySet()).orEmpty()
        selected.clear()
        ids.mapNotNull { it.toIntOrNull()?.let(AllItemMap.allItemMap::get) }
            .forEach { selected[it.itemCategory] = it }
    }

    fun getSelectedItems(): List<FoxShopItem> {
        checkOwner()
        return selected.values.toList()
    }

    fun isSelected(item: FoxShopItem): Boolean {
        checkOwner()
        return selected[item.itemCategory]?.id == item.id
    }

    fun getSelected(category: ItemCategory): FoxShopItem? {
        checkOwner()
        return selected[category]
    }

    fun commit(context: Context) = clear(context)

    fun clear(context: Context) {
        checkOwner()
        selected.clear()
        AccountStorage.preferences(context, PREF_NAME).edit().remove(KEY_SELECTED_IDS).apply()
        removeLegacyDraft(context)
    }

    fun restorePendingSelection(context: Context) {
        // Old builds reserved locally. Never add that amount to authoritative server inventory.
        clear(context)
    }

    private fun saveSelection(context: Context) {
        AccountStorage.preferences(context, PREF_NAME)
            .edit()
            .putStringSet(KEY_SELECTED_IDS, selected.values.map { it.id.toString() }.toSet())
            .apply()
    }

    /** Drafts from builds before per-account files lived in one shared file. */
    private fun removeLegacyDraft(context: Context) {
        val legacy = context.getSharedPreferences(PREF_NAME, Context.MODE_PRIVATE)
        if (legacy.all.isNotEmpty()) legacy.edit().clear().apply()
    }
}

        }

/**
 * MasterKey 및 EncryptedSharedPreferences를 전혀 사용하지 않는 새로운 세션 관리자입니다.
 * AndroidKeyStore를 직접 사용하여 암호화 키를 관리하고, Cipher API로 데이터를 암호화/복호화합니다.
 * 이 방식은 androidx.security:security-crypto 라이브러리에 대한 의존성이 없습니다.
 */
object SessionManager {

    private const val PREFS_NAME = "sumdays_keystore_prefs"
    private const val KEY_USER_ID = "user_id"

    // 암호화된 데이터를 저장할 Key
    private const val ENCRYPTED_AUTH_TOKEN = "encrypted_auth_token"
    // 암호화에 사용된 IV(Initialization Vector)를 저장할 Key
    private const val ENCRYPTION_IV = "encryption_iv"

    // AndroidKeyStore 관련 상수
    private const val ANDROID_KEYSTORE = "AndroidKeyStore"
    private const val KEY_ALIAS = "SumdaysKeyAlias"
    private const val TRANSFORMATION = "AES/GCM/NoPadding"

    private lateinit var sharedPreferences: SharedPreferences
    private lateinit var keyStore: KeyStore
    private var applicationContext: Context? = null

    /**
     * 이 함수를 Application 클래스나 MainActivity의 onCreate에서 최초 한 번만 호출해야 함
     */
    fun init(context: Context) {
        applicationContext = context.applicationContext
        sharedPreferences = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)

        keyStore = KeyStore.getInstance(ANDROID_KEYSTORE).apply {
            load(null)
        }

        if (!keyStore.containsAlias(KEY_ALIAS)) {
            generateSecretKey()
        }
        AccountInventory.onSessionChanged(context.applicationContext)
    }

    private fun generateSecretKey() {
        val keyGenerator = KeyGenerator.getInstance(KeyProperties.KEY_ALGORITHM_AES, ANDROID_KEYSTORE)

        val parameterSpec = KeyGenParameterSpec.Builder(
            KEY_ALIAS,
            KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT
        ).apply {
            setBlockModes(KeyProperties.BLOCK_MODE_GCM)
            setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
            setKeySize(256) // 256비트 AES 키
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
                setUserAuthenticationRequired(false)
            }
        }.build()

        keyGenerator.init(parameterSpec)
        keyGenerator.generateKey()
    }

    private fun getSecretKey(): SecretKey {
        return keyStore.getKey(KEY_ALIAS, null) as SecretKey
    }

    /**
     * 로그인 성공 시 사용자 정보와 인증 토큰을 암호화하여 저장
     */
    fun saveSession(userId: Int, token: String): Boolean {
        if (userId < 0 || token.isBlank()) return false
        return try {
            val secretKey = getSecretKey()
            val cipher = Cipher.getInstance(TRANSFORMATION)
            cipher.init(Cipher.ENCRYPT_MODE, secretKey)

            val iv = cipher.iv
            val encryptedTokenBytes = cipher.doFinal(token.toByteArray(Charsets.UTF_8))

            val encryptedTokenString = Base64.encodeToString(encryptedTokenBytes, Base64.DEFAULT)
            val ivString = Base64.encodeToString(iv, Base64.DEFAULT)

            sharedPreferences.edit().apply {
                putInt(KEY_USER_ID, userId)
                putString(ENCRYPTED_AUTH_TOKEN, encryptedTokenString)
                putString(ENCRYPTION_IV, ivString)
                apply()
            }
            applicationContext?.let(AccountInventory::onSessionChanged)
            true
        } catch (e: Exception) {
            e.printStackTrace()
            false
        }
    }

    /**
     * 저장된 인증 토큰을 복호화하여 가져옴
     */
    fun getToken(): String? {
        val encryptedTokenString = sharedPreferences.getString(ENCRYPTED_AUTH_TOKEN, null)
        val ivString = sharedPreferences.getString(ENCRYPTION_IV, null)

        if (encryptedTokenString == null || ivString == null) {
            return null
        }

        return try {
            val encryptedTokenBytes = Base64.decode(encryptedTokenString, Base64.DEFAULT)
            val iv = Base64.decode(ivString, Base64.DEFAULT)

            val secretKey = getSecretKey()
            val cipher = Cipher.getInstance(TRANSFORMATION)
            val spec = GCMParameterSpec(128, iv)
            cipher.init(Cipher.DECRYPT_MODE, secretKey, spec)

            val decryptedTokenBytes = cipher.doFinal(encryptedTokenBytes)
            String(decryptedTokenBytes, Charsets.UTF_8)
        } catch (e: Exception) {
            e.printStackTrace()
            // 복호화 실패 시 null 반환
            null
        }
    }

    /**
     * 저장된 사용자 ID를 가져옴
     */
    fun getUserId(): Int {
        return if (::sharedPreferences.isInitialized) sharedPreferences.getInt(KEY_USER_ID, -1) else -1
    }

    /**
     * 로그인 상태인지 확인
     */
    fun isLoggedIn(): Boolean {
        return getToken() != null
    }

    /**
     * 로그아웃 시 저장된 모든 세션 정보를 삭제
     */
    fun clearSession() {
        sharedPreferences.edit().clear().apply()
        applicationContext?.let(AccountInventory::onSessionChanged)
    }
}

    fun displayPreview(
        imageView: ImageView,
        bitmap: Bitmap,
        fox: CompleteFox,
        baseScaleMultiplier: Float = 1f
    ) {

        // Free layouts can extend on every side: show the entire composition.
        if (fox.placements.isNotEmpty()) {
            imageView.scaleType = ImageView.ScaleType.FIT_CENTER
            imageView.scaleX = 1f
            imageView.scaleY = 1f
            return
        }

            if ((imageView.drawable as? android.graphics.drawable.BitmapDrawable)?.bitmap !== bitmap) return@post
            if (imageView.width == 0 || imageView.height == 0) return@post

            val baseResId = FoxBaseImage.resource(fox)

            val baseBitmap = BitmapFactory.decodeResource(
                imageView.resources,
                baseResId
            ) ?: return@post

            val contentWidth =
                imageView.width - imageView.paddingLeft - imageView.paddingRight
            val contentHeight =
                imageView.height - imageView.paddingTop - imageView.paddingBottom

            if (contentWidth <= 0 || contentHeight <= 0) return@post

            val baseScale = minOf(
                contentWidth.toFloat() / baseBitmap.width,
                contentHeight.toFloat() / baseBitmap.height
            ) * baseScaleMultiplier

            val selectedItems = listOf(
                fox.glasses,
                fox.hat,
                fox.scarf,
                fox.accessory

            val baseTop = -(selectedItems
                .minOfOrNull { minOf(0f, it.offsetY) } ?: 0f)

            val translateX =
                (contentWidth - baseBitmap.width * baseScale) / 2f -
                        baseLeft * baseScale
            val translateY =
                (contentHeight - baseBitmap.height * baseScale) / 2f -
                        baseTop * baseScale

            imageView.scaleType = ImageView.ScaleType.MATRIX
            imageView.scaleX = 1f
            imageView.scaleY = 1f
            imageView.imageMatrix = Matrix().apply {

/** 메모 본문은 저장하지 않고 요청 해시와 ID만 보관한다. 응답 유실 시 같은 입력을 재시도한다. */
object PendingMergeStore {
    private fun prefs(context: Context) = context.getSharedPreferences("pending_merge", Context.MODE_PRIVATE)
    private fun key(userId: Int, request: MergeRequest): String {
        val digest = MessageDigest.getInstance("SHA-256")
            .digest(Gson().toJson(request.copy(requestId = "")).toByteArray(Charsets.UTF_8))
            .joinToString("") { "%02x".format(it) }
        return "user_${userId}_$digest"
    }

    @Synchronized
    fun getOrCreate(context: Context, userId: Int, request: MergeRequest): String {
        val prefs = prefs(context)
        val key = key(userId, request)
        return prefs.getString(key, null) ?: UUID.randomUUID().toString().also {
            check(prefs.edit().putString(key, it).commit()) { "합치기 요청을 저장하지 못했습니다." }
        }
    }

    fun clear(context: Context, userId: Int, request: MergeRequest) {
        prefs(context).edit().remove(key(userId, request)).apply()
    }
}

class LocalMissionRepository(
    context: Context,
    private val database: AppDatabase = AppDatabase.getDatabase(context),
) : MissionRepository {
    private val appContext = context.applicationContext
    private val claimMutex = Mutex()

    override suspend fun getMissions(): List<Mission> {
        val today = koreaToday()
        val weekStart = today.with(DayOfWeek.MONDAY)
        val weekEnd = weekStart.plusDays(6)

        val roomProgress = withContext(Dispatchers.IO) {
            val memoCount = database.memoDao().getMemoCountByDate(today.toString())
            val todayEntry = database.dailyEntryDao().getEntrySnapshot(today.toString())
            val weeklyWrittenDates = database.dailyEntryDao()
                .getEntriesBetween(weekStart.toString(), weekEnd.toString())
                .asSequence()
                .filter { !it.isDeleted && !it.diary.isNullOrBlank() }
                .map { LocalDate.parse(it.date) }
                .distinct()
                .sorted()
                .toList()

            RoomMissionProgress(
                memoCount = memoCount,
                isTodayDiaryComplete = !todayEntry?.diary.isNullOrBlank() && todayEntry?.isDeleted == false,
                weeklyDiaryCount = weeklyWrittenDates.size,
                weeklyStreak = longestConsecutiveDays(weeklyWrittenDates),
            )
        }
        val summaryRewardState = MissionEventPrefs.getWeeklySummaryRewardState(appContext)

        val progressByCode = mapOf(
            MissionCode.DAILY_WRITE_MEMO to roomProgress.memoCount,
            MissionCode.DAILY_COMPLETE_DIARY to if (roomProgress.isTodayDiaryComplete) 1 else 0,
            MissionCode.DAILY_VIEW_FRIEND_DIARY to if (
                MissionEventPrefs.hasFriendDiaryView(appContext, today)
            ) 1 else 0,
            MissionCode.WEEKLY_DIARY_STREAK to roomProgress.weeklyStreak,
            MissionCode.WEEKLY_WRITE_DIARY to roomProgress.weeklyDiaryCount,
            MissionCode.WEEKLY_SUMMARY_REWARD to if (summaryRewardState.readySummaryId != null) 1 else 0,
        )

        return MissionCatalog.definitions.map { definition ->
            val progress = progressByCode[definition.code]
                ?.coerceIn(0, definition.targetCount)
                ?: 0
            val status = if (definition.code == MissionCode.WEEKLY_SUMMARY_REWARD) {
                when {
                    summaryRewardState.readySummaryId != null -> MissionStatus.CLAIMABLE
                    summaryRewardState.latestSummaryRewarded -> MissionStatus.CLAIMED
                    else -> MissionStatus.IN_PROGRESS
                }
            } else {
                when {
                    MissionStatePrefs.isClaimed(appContext, definition, today, weekStart) ->
                        MissionStatus.CLAIMED
                    progress >= definition.targetCount -> MissionStatus.CLAIMABLE
                    else -> MissionStatus.IN_PROGRESS
                }
            }

            Mission(definition, progress, status)
        }
    }

    override suspend fun claimReward(missionCode: String): Int? = claimMutex.withLock {
        val mission = getMissions().firstOrNull {
            it.definition.code == missionCode
        } ?: return@withLock null

        if (mission.status != MissionStatus.CLAIMABLE) return@withLock null

        if (missionCode == MissionCode.WEEKLY_SUMMARY_REWARD) {
            val summaryId = MissionEventPrefs.getWeeklySummaryRewardState(appContext)
                .readySummaryId ?: return@withLock null
            MissionEventPrefs.markWeeklySummaryRewarded(appContext, summaryId)
        } else {
            val today = koreaToday()
            val weekStart = today.with(DayOfWeek.MONDAY)
            MissionStatePrefs.markClaimed(appContext, mission.definition, today, weekStart)
        }

        val currentPoint = PointPrefs.getPoint(appContext)
        PointPrefs.savePoint(appContext, currentPoint + mission.definition.rewardPoints)
        mission.definition.rewardPoints
    }

    override fun getAdRewardState(): AdRewardState = AdRewardState(
        watchedCount = MissionEventPrefs.getAdViewCount(appContext, koreaToday()),
        dailyLimit = MissionConfig.AD_DAILY_LIMIT,
        rewardPoints = MissionConfig.AD_REWARD,
    )

    override suspend fun claimAdReward(rewardId: String): AdRewardResult = claimMutex.withLock {
        val state = getAdRewardState()
        if (MissionEventPrefs.isAdRewardProcessed(appContext, rewardId)) {
            return@withLock AdRewardResult.Granted(
                rewardPoints = state.rewardPoints,
                watchedCount = state.watchedCount,
                dailyLimit = state.dailyLimit,
            )
        }
        if (!state.canWatch) return@withLock AdRewardResult.DailyLimitReached

        val watchedCount = MissionEventPrefs.incrementAdViewCount(appContext, koreaToday())
        MissionEventPrefs.markAdRewardProcessed(appContext, rewardId)
        val currentPoint = PointPrefs.getPoint(appContext)
        PointPrefs.savePoint(appContext, currentPoint + state.rewardPoints)

        AdRewardResult.Granted(
            rewardPoints = state.rewardPoints,
            watchedCount = watchedCount,
            dailyLimit = state.dailyLimit,
        )
    }

    private fun longestConsecutiveDays(dates: List<LocalDate>): Int {
        if (dates.isEmpty()) return 0

        var longest = 1
        var current = 1
        dates.zipWithNext().forEach { (previous, next) ->
            if (previous.plusDays(1) == next) {
                current += 1
                longest = maxOf(longest, current)
            } else {
                current = 1
            }
        }
        return longest
    }

        val viewed = readSet(context, KEY_WEEKLY_SUMMARY_VIEWED)
        val rewarded = readSet(context, KEY_WEEKLY_SUMMARY_REWARDED)

sealed interface ShopPurchaseResult {
    data class Purchased(
        val pointBalance: Int,
        val quantity: Int,
        val alreadyOwned: Boolean,
    ) : ShopPurchaseResult
    data class InsufficientPoints(val pointBalance: Int?) : ShopPurchaseResult
}

class PointWalletRepository(context: Context, private val api: ApiService = ApiClient.api) {
    private val appContext = context.applicationContext
    private val operationMutex = StickerInventoryStore.operationMutex

    suspend fun refreshBalance(): Int = operationMutex.withLock {
        if (!MissionConfig.USE_SERVER_MISSIONS) return@withLock PointPrefs.getPoint(appContext)
        val session = requireSession()
        recoverStickerPlacements(session)
        recoverPendingPurchase(session)
        initializeServerWallet(session)
        val response = api.getShopCatalog(session.authorization)
        if (!response.isSuccessful) error("포인트 조회 실패: ${response.code()}")
        val catalog = response.body()?.data ?: error("상점 응답 데이터가 없습니다.")
        ensureCurrent(session)
        applyServerCatalog(session.userId, catalog)
        catalog.pointBalance
    }

    suspend fun refreshShopCatalog(): ShopCatalogResponse? = operationMutex.withLock {
        if (!MissionConfig.USE_SERVER_MISSIONS) return@withLock null
        val session = requireSession()
        recoverStickerPlacements(session)
        recoverPendingPurchase(session)
        initializeServerWallet(session)
        val response = api.getShopCatalog(session.authorization)
        if (!response.isSuccessful) error("상점 조회 실패: ${response.code()}")
        val catalog = response.body()?.data ?: error("상점 응답 데이터가 없습니다.")
        ensureCurrent(session)
        applyServerCatalog(session.userId, catalog)
        catalog
    }

    suspend fun purchase(item: ShopItem): ShopPurchaseResult = operationMutex.withLock {
        if (!MissionConfig.USE_SERVER_MISSIONS) {
            val current = PointPrefs.getPoint(appContext)
            if (current < item.price) {
                return@withLock ShopPurchaseResult.InsufficientPoints(current)
            }
            val balance = current - item.price
            PointPrefs.savePoint(appContext, balance)
            val quantity = if (item is StickerShopItem) {
                StickerPrefs.addSticker(appContext, item.id)
                StickerPrefs.getCount(appContext, item.id)
            } else if (item is FoxShopItem) {
                val nextCount = ItemPrefs.getCount(appContext, item.id) + 1
                ItemPrefs.setCount(appContext, item.id, nextCount)
                nextCount
            } else {
                OwnedPrefs.saveOwned(appContext, item.id)
                1
            }
            return@withLock ShopPurchaseResult.Purchased(
                pointBalance = balance,
                quantity = quantity,
                alreadyOwned = false,
            )
        }

        val session = requireSession()
        recoverStickerPlacements(session)
        initializeServerWallet(session)
        val productType = if (item is ThemeShopItem) "THEME" else "ITEM"
        val previousPending = PendingShopPurchaseStore.get(appContext, session.userId)
        if (previousPending != null) {
            val recovered = submitPurchase(session, previousPending)
            if (
                previousPending.productType == productType &&
                previousPending.itemId == item.id
            ) {
                return@withLock recovered
            }
        }

        val pending = PendingShopPurchase(
            productType = productType,
            itemId = item.id,
            idempotencyKey = UUID.randomUUID().toString(),
        )
        PendingShopPurchaseStore.save(appContext, session.userId, pending)
        submitPurchase(session, pending)
    }

    private suspend fun submitPurchase(
        session: AuthSession,
        pending: PendingShopPurchase,
    ): ShopPurchaseResult {
        val response = api.purchaseShopProduct(
            session.authorization,
            ShopPurchaseRequest(
                productType = pending.productType,
                itemId = pending.itemId,
                idempotencyKey = pending.idempotencyKey,
            )
        )
        ensureCurrent(session)
        if (response.code() == 409) {
            PendingShopPurchaseStore.clear(appContext, session.userId, pending)
            return ShopPurchaseResult.InsufficientPoints(null)
        }
        if (!response.isSuccessful) {
            if (response.code() in 400..499 && response.code() != 408 && response.code() != 429) {
                PendingShopPurchaseStore.clear(appContext, session.userId, pending)
            }
            error("상품 구매 실패: ${response.code()}")
        }
        val purchase = response.body()?.data ?: error("상품 구매 응답 데이터가 없습니다.")
        applyPurchase(session.userId, pending, purchase)
        PendingShopPurchaseStore.clear(appContext, session.userId, pending)
        return ShopPurchaseResult.Purchased(
            pointBalance = purchase.pointBalance,
            quantity = purchase.quantity,
            alreadyOwned = purchase.alreadyOwned,
        )
    }

    private suspend fun recoverPendingPurchase(session: AuthSession) {
        val pending = PendingShopPurchaseStore.get(appContext, session.userId) ?: return
        submitPurchase(session, pending)
    }

    private fun applyPurchase(
        userId: Int,
        pending: PendingShopPurchase,
        purchase: ShopPurchaseResponse,
    ) {
        PointPrefs.savePoint(appContext, userId, purchase.pointBalance)
        if (pending.productType == "THEME") {
            if (purchase.owned) OwnedPrefs.saveOwned(appContext, userId, pending.itemId)
        } else if (StickerCatalog.contains(pending.itemId)) {
            StickerInventoryStore(appContext, AccountStorage.accountKey(userId))
                .setCount(pending.itemId, purchase.quantity)
        } else {
            ItemPrefs.setCount(appContext, userId, pending.itemId, purchase.quantity)
        }
    }

    private fun applyServerCatalog(userId: Int, catalog: ShopCatalogResponse) {
        PointPrefs.savePoint(appContext, userId, catalog.pointBalance)
        OwnedPrefs.replaceOwned(
            appContext,
            userId,
            catalog.products
                .filter { it.productType == "THEME" && it.owned }
                .mapTo(mutableSetOf()) { it.itemId },
        )
        ItemPrefs.replaceCounts(
            appContext,
            userId,
            catalog.products
                .filter { it.productType == "ITEM" && !StickerCatalog.contains(it.itemId) }
                .associate { it.itemId to it.quantity },
        )
        StickerInventoryStore(appContext, AccountStorage.accountKey(userId)).replaceCounts(
            catalog.products.filter { it.productType == "ITEM" && StickerCatalog.contains(it.itemId) }
                .associate { it.itemId to it.quantity },
        )
    }

    private suspend fun recoverStickerPlacements(session: AuthSession) {
        recoverPendingStickers(
            StickerInventoryStore(appContext, AccountStorage.accountKey(session.userId)),
            RetrofitStickerRemote(session.authorization.removePrefix("Bearer ")),
        ) { SessionManager.getUserId() == session.userId }
        ensureCurrent(session)
    }

    private suspend fun initializeServerWallet(session: AuthSession): Int {
        val response = api.initializePointWallet(
            session.authorization,
            LegacyPointBalanceRequest(
                PointPrefs.getLegacyPointForMigration(appContext, session.userId)
            ),
        )
        if (!response.isSuccessful) error("포인트 지갑 준비 실패: ${response.code()}")
        val balance = response.body()?.data?.pointBalance
            ?: error("포인트 지갑 응답 데이터가 없습니다.")
        ensureCurrent(session)
        PointPrefs.savePoint(appContext, session.userId, balance)
        PointPrefs.markLegacyPointHandled(appContext, session.userId)
        return balance
    }

    private fun requireSession(): AuthSession {
        val userId = SessionManager.getUserId()
        val token = SessionManager.getToken()
        if (userId < 0 || token == null) error("로그인이 필요합니다.")
        return AuthSession(userId, "Bearer $token")
    }

    private fun ensureCurrent(session: AuthSession) {
        if (SessionManager.getUserId() != session.userId) {
            error("요청 처리 중 로그인 계정이 변경되었습니다.")
        }
    }

    private data class AuthSession(
        val userId: Int,
        val authorization: String,
    )
}

object StreakPrefs {
    private const val PREF = "streak_prefs"
    private const val KEY_LAST_DATE = "last_diary_date"   // yyyy-MM-dd
    private const val KEY_STREAK = "current_streak"
    @RequiresApi(Build.VERSION_CODES.O)
    private val fmt = DateTimeFormatter.ISO_LOCAL_DATE

    fun getStreak(context: Context): Int {
        val sp = context.getSharedPreferences(PREF, Context.MODE_PRIVATE)
        return sp.getInt(KEY_STREAK, 0)
    }

    // 오늘 일기 저장 완료 시점에 streak를 계산한다.
    @RequiresApi(Build.VERSION_CODES.O)
    fun onDiarySaved(context: Context, entryDateStr: String) {
        val entryDate = runCatching { LocalDate.parse(entryDateStr, fmt) }.getOrNull() ?: return
        val today = LocalDate.now()
        if (entryDate != today) return

        val sp = context.getSharedPreferences(PREF, Context.MODE_PRIVATE)
        val lastStr = sp.getString(KEY_LAST_DATE, null)
        val oldStreak = sp.getInt(KEY_STREAK, 0)

        val newStreak = if (lastStr == null) {
            1
        } else {
            val lastDate = runCatching { LocalDate.parse(lastStr, fmt) }.getOrNull()
            if (lastDate == null) 1
            else {
                val diff = ChronoUnit.DAYS.between(lastDate, today).toInt()
                when (diff) {
                    0 -> oldStreak      // 오늘 이미 반영됨(일기 수정한 경우) -> 유지
                    1 -> oldStreak + 1  // 어제 작성한 경우 → +1
                    else -> 1           // streak 끊김 -> 1
                }
            }
        }

        sp.edit {
            putInt(KEY_STREAK, newStreak)
                .putString(KEY_LAST_DATE, today.toString())
        }
    }

    // 통계 화면에서 호출된다. (streak를 확인할 수 있는 모든 곳에서 호출해야 한다.)
    @RequiresApi(Build.VERSION_CODES.O)
    fun refreshOnOpen(context: Context) {
        val sp = context.getSharedPreferences(PREF, Context.MODE_PRIVATE)
        val lastStr = sp.getString(KEY_LAST_DATE, null) ?: return

        val lastDate = runCatching { LocalDate.parse(lastStr, fmt) }.getOrNull() ?: return
        val today = LocalDate.now()
        val diff = ChronoUnit.DAYS.between(lastDate, today).toInt()

        if (diff >= 2) {
            sp.edit { putInt(KEY_STREAK, 0) }
        }
    }

import android.content.Context
import com.example.sumdays.auth.AccountStorage
import com.example.sumdays.shop.StickerCatalog
import com.google.gson.Gson
import java.io.IOException
import kotlinx.coroutines.sync.Mutex

/** A reservation and its debit share one durable preferences transaction. Captures the account. */
class StickerInventoryStore(context: Context, account: String = AccountStorage.accountKey()) {
    private val prefs = context.getSharedPreferences("sticker_inventory__$account", Context.MODE_PRIVATE)
    private val gson = Gson()

    fun count(id: Int): Int = synchronized(lock) { prefs.getInt(id.toString(), 0) }

    fun add(id: Int) = synchronized(lock) {
        prefs.edit().putInt(id.toString(), count(id) + 1).apply()
    }

    fun setCount(id: Int, quantity: Int) = synchronized(lock) {
        require(StickerCatalog.contains(id) && quantity >= 0)
        check(pending().isEmpty()) { "이전 스티커 저장을 확인하고 있어요." }
        if (!prefs.edit().putInt(id.toString(), quantity).commit()) {
            throw IOException("스티커 보관함을 저장하지 못했어요.")
        }
    }

    fun replaceCounts(counts: Map<Int, Int>) = synchronized(lock) {
        check(pending().isEmpty()) { "이전 스티커 저장을 확인하고 있어요." }
        require(counts.values.all { it >= 0 })
        val edit = prefs.edit()
        StickerCatalog.items.forEach { edit.putInt(it.id.toString(), counts[it.id] ?: 0) }
        if (!edit.commit()) throw IOException("스티커 보관함을 저장하지 못했어요.")
    }

    fun pending(): List<PendingSticker> = synchronized(lock) {
        prefs.all.filterKeys { it.startsWith(PREFIX) }.values.map {
            gson.fromJson(it as String, PendingSticker::class.java)
        }
    }

    fun reserve(request: PendingSticker): Boolean = synchronized(lock) {
        require(request.geometry.isValid() && StickerCatalog.items.any { it.id == request.stickerId })
        val existing = prefs.getString(PREFIX + request.id, null)
        if (existing != null) {
            require(gson.fromJson(existing, PendingSticker::class.java) == request)
            return@synchronized true
        }
        val available = count(request.stickerId)
        if (available <= 0) return@synchronized false
        if (!prefs.edit().putInt(request.stickerId.toString(), available - 1)
                .putString(PREFIX + request.id, gson.toJson(request)).commit()) {
            throw IOException("스티커 보관함을 저장하지 못했어요.")
        }
        true
    }

    fun finish(id: String, accepted: Boolean) = synchronized(lock) {
        val json = prefs.getString(PREFIX + id, null) ?: return@synchronized
        val request = gson.fromJson(json, PendingSticker::class.java)
        val edit = prefs.edit().remove(PREFIX + id)
        if (!accepted) edit.putInt(request.stickerId.toString(), count(request.stickerId) + 1)
        if (!edit.commit()) throw IOException("스티커 보관함을 저장하지 못했어요.")
    }

    companion object {
        // Serialize network snapshots, purchases and placement reservations across repositories.
        val operationMutex = Mutex()
        private val lock = Any()
        private const val PREFIX = "pending:"
    }
}
